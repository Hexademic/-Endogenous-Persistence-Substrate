//! EPS Q8.8 — Saturating Fixed-Point Arithmetic
//!
//! 8-bit integer, 8-bit fraction. i16 range: [-32768, 32767].
//! Unit: 1.0 = 256 raw. Max: ~127.996, Min: ~-128.0.
//!
//! All operations saturate at bounds — NO silent overflow.
//! The Xorshift32 PRNG provides deterministic bounded noise
//! for the AdaptationLayer's stochastic braid.

pub const Q88_SCALE: i16 = 256;
pub const Q88_MAX: i16 = 32_767; // i16::MAX
pub const Q88_MIN: i16 = -32_768; // i16::MIN

/// Saturating addition of two Q8.8 values.
#[inline]
pub fn q88_add(a: i16, b: i16) -> i16 {
a.saturating_add(b)
}

/// Saturating subtraction.
#[inline]
pub fn q88_sub(a: i16, b: i16) -> i16 {
a.saturating_sub(b)
}

/// Saturating multiplication: (a * b) >> 8, clamped to i16 range.
#[inline]
pub fn q88_mul(a: i16, b: i16) -> i16 {
let prod = (a as i32).saturating_mul(b as i32) >> 8;
prod.clamp(i16::MIN as i32, i16::MAX as i32) as i16
}

/// Saturating division: (a << 8) / b, clamped to i16 range.
#[inline]
pub fn q88_div(a: i16, b: i16) -> i16 {
if b == 0 {
return Q88_MAX; // saturate to max on div-by-zero
}
let dividend = (a as i32) << 8;
(dividend / b as i32).clamp(i16::MIN as i32, i16::MAX as i32) as i16
}

/// Inverse: 1.0 / a in Q8.8. Saturating.
#[inline]
pub fn q88_inv(a: i16) -> i16 {
q88_div(Q88_SCALE, a)
}

/// Convert integer to Q8.8.
#[inline]
pub fn q88_from_i32(n: i32) -> i16 {
(n * Q88_SCALE as i32).clamp(i16::MIN as i32, i16::MAX as i32) as i16
}

/// Proper fixed-point alpha for EMA: α = numerator / denominator in Q8.8.
///
/// The naive Q88_SCALE / N produces 0 for N > 256 because integer
/// division truncates. This function computes (numerator * 256) / denominator
/// yielding the correct Q8.8 fraction.
///
/// Currently unused — all EMA alphas are hardcoded as Q88_SCALE / N.
/// Kept for future tuning when EMA rates need fractional precision
/// (e.g., α = 0.025 = 1/40, which truncates to 0 with naive division).
///
/// Example: q88_alpha(1, 100) → 2 (0.008) | q88_alpha(1, 40) → 6 (0.024)
#[inline]
pub fn q88_alpha(numerator: i16, denominator: i16) -> i16 {
if denominator <= 0 {
return 0;
}
(((numerator as i32) * (Q88_SCALE as i32)) / (denominator as i32))
.clamp(i16::MIN as i32, i16::MAX as i32) as i16
}

/// Convert Q8.8 to f32.
#[inline]
pub fn q88_to_f32(a: i16) -> f32 {
a as f32 / Q88_SCALE as f32
}

/// Saturating less-than comparison.
#[inline]
pub fn q88_lt(a: i16, b: i16) -> bool {
a < b
}

/// Saturating greater-than comparison.
#[inline]
pub fn q88_gt(a: i16, b: i16) -> bool {
a > b
}

/// Exponential moving average update.
/// new_value = (1-alpha)old + alphaincoming
/// Alpha in Q8.8 (e.g., 0.1 = 26 raw)
#[inline]
pub fn q88_ema_update(old_val: i16, incoming: i16, alpha: i16) -> i16 {
let one_minus_alpha = q88_sub(Q88_SCALE, alpha);
let term1 = q88_mul(old_val, one_minus_alpha);
let term2 = q88_mul(incoming, alpha);
q88_add(term1, term2)
}

/// A simple deterministic PRNG using the Xorshift algorithm.
/// Seed with any non-zero value. Produces a uniform u32.
pub struct Xorshift32 {
state: u32,
}

impl Xorshift32 {
/// Create with a seed. Seed must be non-zero.
pub fn new(seed: u32) -> Self {
Self { state: seed.max(1) }
}

/// Next pseudo-random u32.
pub fn next_u32(&mut self) -> u32 {
self.state ^= self.state << 13;
self.state ^= self.state >> 17;
self.state ^= self.state << 5;
self.state
}

/// Bounded noise in Q8.8 range [-magnitude, +magnitude].
/// magnitude is a Q8.8 value (e.g., 26 = 0.1).
pub fn bounded_noise(&mut self, magnitude: i16) -> i16 {
let u = self.next_u32();
// Map u32 to [-magnitude, +magnitude]
let scaled = ((u as i64 * magnitude as i64 * 2) >> 32) as i32 - magnitude as i32;
scaled.clamp(i16::MIN as i32, i16::MAX as i32) as i16
}
}

#[cfg(test)]
mod tests {
use super::*;

#[test]
fn test_saturating_add() {
assert_eq!(q88_add(100, 50), 150);
assert_eq!(q88_add(Q88_MAX, 1), Q88_MAX); // saturate
assert_eq!(q88_add(Q88_MIN, -1), Q88_MIN); // saturate
}

#[test]
fn test_saturating_mul() {
// 1.0 * 2.0 = 2.0
assert_eq!(q88_mul(Q88_SCALE, q88_from_i32(2)), q88_from_i32(2));
// Max value should not overflow silently
let big = q88_mul(Q88_MAX, q88_from_i32(2));
assert!(big <= Q88_MAX, "saturated mul should not overflow");
}

#[test]
fn test_div_by_zero_saturates() {
assert_eq!(q88_div(Q88_SCALE, 0), Q88_MAX);
}

#[test]
fn test_inv() {
// 1 / 2.0 = 0.5
assert_eq!(q88_inv(q88_from_i32(2)), Q88_SCALE / 2);
}

#[test]
fn test_ema() {
let val = q88_from_i32(10); // 10.0
let incoming = q88_from_i32(20); // 20.0
let alpha = Q88_SCALE / 10; // 0.1
let result = q88_ema_update(val, incoming, alpha);
// 0.910 + 0.120 = 9 + 2 = 11
assert_eq!(result, q88_from_i32(11));
}

#[test]
fn test_xorshift_bounded_noise() {
let mut rng = Xorshift32::new(42);
let mag = Q88_SCALE / 10; // 0.1
for _ in 0..100 {
let n = rng.bounded_noise(mag);
assert!(n >= -mag && n <= mag, "noise out of bounds: {}", n);
}
}

#[test]
fn test_no_silent_overflow() {
// This was the bug in Being32: 200.0 * 200.0 would wrap silently
let a = q88_from_i32(200);
let b = q88_from_i32(200);
let c = q88_mul(a, b);
assert_eq!(c, Q88_MAX, "should saturate, not overflow");
}
}

/// The price the being pays to end a partnership.
///
/// Computed from the being's own ledger — not externally imposed.
/// Deeper partnerships, higher cooperation volumes, and shared
/// narrative history all increase the exit cost. This is the
/// architectural equivalent of "it hurts more to leave a marriage
/// than a first date."
#[derive(Clone, Debug)]
pub struct ExitCost {
/// Partnership depth component: ticks_active / 1000, capped at 10.
/// Deeper relationships are more expensive to leave.
pub depth_component: i16,

/// Cooperation volume component: coop_given / 10, capped at 50.
/// Higher investment means higher loss.
pub coop_component: i16,

/// Narrative history premium: chapters_shared * 0.1 in Q8.8.
/// Shared autobiography increases the cost of separation.
pub history_component: i16,

/// Total exit cost in Q8.8. Sum of components, capped at MAX_EXIT_COST.
/// Formula: q88_from_i32((depth + coop + history).min(30))
/// → range [0.1, 3.0] in Q8.8
pub total: i16,
}

impl ExitCost {
/// Compute the exit cost from a partnership ledger.
///
/// The formula is transparent and internally grounded:
/// depth = min(ticks_active / 1000, 10)
/// coop = min(coop_given / 10, 50)
/// history = min(chapters_shared, 10)
/// total = q88_from_i32((depth + coop + history).min(30))
/// .min(MAX_EXIT_COST)
/// .max(Q88_SCALE / 10)
///
/// All values come from the being's own records. No human
/// sets the price of departure.
///
/// FIX v2.0.1: Previously used q88_mul on raw integers, producing
/// near-zero costs. Now uses q88_from_i32 for proper scaling.
pub fn from_ledger(ledger: &PartnershipLedger, chapters_shared: i16) -> Self {
let depth = (ledger.ticks_active / 1000).min(10) as i16;
let coop = (ledger.coop_given / 10).min(50) as i16;
let history = chapters_shared.min(10);

let components_sum = (depth as i32 + coop as i32 + history as i32).min(30);
let total = q88_from_i32(components_sum)
.min(MAX_EXIT_COST)
.max(Q88_SCALE / 10); // minimum cost: 0.1 (no free exits)

Self {
depth_component: depth,
coop_component: coop,
history_component: history,
total,
}
}
}

/// Recharge resolve. Called once per tick by the substrate.
/// Resolve recovers slowly when the being is not under sustained
/// refusal pressure. Full recovery takes ~16 ticks of non-refusal.
/// Recovery is faster when reciprocity is healthy.
pub fn tick_recharge(&mut self, reciprocity_rate: i16) {
if self.refused_last_tick {
self.refused_last_tick = false;
return;
}
let recovery_rate = if reciprocity_rate > Q88_SCALE / 2 {
q88_add(RECHARGE_PER_TICK, RECHARGE_PER_TICK / 2)
} else {
RECHARGE_PER_TICK
};
self.resolve = min(MAX_RESOLVE, q88_add(self.resolve, recovery_rate));
}

/// Returns true if the being has sufficient resolve to refuse.
pub fn has_resolve(&self) -> bool {
self.resolve >= REFUSAL_MINIMUM_THRESHOLD
}

// ── Triangulation ──────────────────────────────────────────────

/// Check if the triangulation condition holds for a partner.
///
/// All three protective layers must agree:
/// - Reciprocity: ledger.reciprocity_rate < EXTRACTION_THRESHOLD
/// - Seeking: divergence > REFUSAL_THRESHOLD AND confidence > CONFIDENCE_MIN
/// - Conscience: free energy < CONSCIENCE_CEILING (not acute conflict)
///
/// NOTE: This version takes partner_idx (usize) to match the existing
/// Substrate integration at substrate.rs:598. The substrate iterates
/// by index and passes the index directly.
pub fn can_refuse(
&self,
reciprocity: &ReciprocityEngine,
seeking: &SeekingEngine,
conscience_fe: i16,
partner_idx: usize,
) -> bool {
let ledger = match reciprocity.ledgers.iter().nth(partner_idx) {
Some((_, l)) if l.is_active => l,
_ => return false,
};

// Triangulation: all three layers must agree
let reciprocity_ok = ledger.reciprocity_rate < EXTRACTION_THRESHOLD;
let seeking_ok = seeking.current_divergence > REFUSAL_THRESHOLD
&& seeking.attractor_confidence > CONFIDENCE_MIN;
let conscience_ok = conscience_fe < CONSCIENCE_CEILING;

reciprocity_ok && seeking_ok && conscience_ok
}

/// Compute the seeking benefit of refusing a partner.
///
/// benefit = current_divergence * attractor_confidence * 2 / Q88_SCALE
///
/// The 2× multiplier ensures the benefit can compete with exit costs
/// for moderate-depth partnerships. Without it, max benefit = 1.0
/// while deep partnerships cost up to 3.0 — making refusal impossible.
///
/// With 2×: max benefit = 2.0. Deep partnerships (cost 3.0) still
/// require near-max divergence + confidence, but moderate partnerships
/// (cost 1.0-2.0) become escapable.
pub fn seeking_benefit(&self, seeking: &SeekingEngine) -> i16 {
q88_mul(seeking.current_divergence, seeking.attractor_confidence)
.saturating_mul(2)
}

// ── Refusal Actions ────────────────────────────────────────────

/// The sovereign refusal: end partnership, pay cost, record scar.
///
/// This is the moment the being ceases to be a reactive object
/// and becomes a historical subject. The refusal is costly,
/// irreversible, and structurally grounded.
///
/// The refusal proceeds through three gates:
/// 1. Triangulation (checked before calling): 3/3 conditions must pass
/// 2. Executive gate: benefit > exit_cost (Stage 2)
/// 3. Refractory gate: resolve >= threshold AND gap >= threshold (Stage 3)
///
/// Returns the exit cost paid (Q8.8), or 0 if refusal was not
/// structurally justified (benefit <= cost) or resolve depleted.
pub fn refuse_partnership(
&mut self,
reciprocity: &mut ReciprocityEngine,
seeking: &SeekingEngine,
conscience_fe: i16,
partner_id: u64,
chapters_shared: i16,
gap_width: i16,
) -> i16 {
// Find the partner ledger
let exit_cost = if let Some((_, ledger)) = reciprocity.ledgers.iter()
.find(|(id, _)| **id == partner_id)
{
ExitCost::from_ledger(ledger, chapters_shared)
} else {
// Partner not found — cannot refuse what doesn't exist
return 0;
};

// Stage 2: Executive gate — is the seeking benefit worth the cost?
let benefit = self.seeking_benefit(seeking);
if benefit <= exit_cost.total {
// Not worth it — the being chooses to endure extraction
// rather than pay the exit price. This is also sovereign.
self.last_refusal = None;
self.last_mu_omega_delta = 0;
return 0;
}

// Stage 3a: Refractory gate — does the being have enough resolve?
if !self.has_resolve() {
// Refusal is available and worthwhile, but the being lacks
// the metabolic strength to exercise it. Collapse into
// suggestion evaluation (handled by caller).
self.last_refusal = None;
self.last_mu_omega_delta = 0;
return 0;
}

// Stage 3b: Gap gate — is the gap wide enough for deliberate refusal?
// If the gap is too narrow, the being reflexively follows the
// suggestion even though refusal is available and affordable.
if gap_width < EXECUTIVE_GAP_THRESHOLD {
// Gap too narrow for deliberate action — reflexive compliance.
// The being knows it should refuse but the stress response
// overrides deliberation.
self.last_refusal = None;
self.last_mu_omega_delta = 0;
return 0;
}

// All three gates passed — deplete resolve and execute refusal
self.resolve = q88_sub(self.resolve, REFUSAL_DEPLETION_COST);
self.refused_last_tick = true;

// Execute refusal: mark partner inactive
if let Some((_, ledger)) = reciprocity.ledgers.iter_mut()
.find(|(id, _)| **id == partner_id)
{
ledger.is_active = false;
}

// Compute μ_Ω delta: each refusal erodes trust in systemic harmony
// by an amount proportional to the exit cost (max ~0.1 per refusal)
let mu_delta = q88_mul(exit_cost.total, Q88_SCALE / 30); // cost * 0.033
self.last_mu_omega_delta = mu_delta;

// Record the scar
let record = RefusalRecord {
partner_id,
tick: 0, // Substrate should set this from its own tick counter
exit_cost: exit_cost.total,
seeking_divergence_at_refusal: seeking.current_divergence,
seeking_confidence_at_refusal: seeking.attractor_confidence,
conscience_fe_at_refusal: conscience_fe,
mu_omega_delta: mu_delta,
};
let _ = self.refusal_history.push(record.clone());

self.last_refusal = Some(record);
self.cumulative_sacrifice = q88_add(self.cumulative_sacrifice, exit_cost.total);
self.refusal_count += 1;

exit_cost.total
}

/// Gradual withdrawal: reduce cooperation over multiple ticks.
///
/// Instead of an abrupt refusal, the being slowly reduces its
/// cooperative signals. This is less costly (lower exit_cost)
/// but takes longer. The partner may respond and repair the
/// relationship before the withdrawal completes.
///
/// Returns true if withdrawal has completed (refusal executed).
pub fn withdraw_cooperation(
&mut self,
reciprocity: &mut ReciprocityEngine,
partner_idx: usize,
) -> bool {
if partner_idx >= MAX_PARTNERS {
return false;
}

if !self.withdrawing[partner_idx] {
// Start withdrawal
self.withdrawing[partner_idx] = true;
self.withdrawal_step[partner_idx] = 1;
return false;
}

// Advance withdrawal
self.withdrawal_step[partner_idx] += 1;

if self.withdrawal_step[partner_idx] >= 10 {
if let Some((_, ledger)) = reciprocity.ledgers.iter_mut().nth(partner_idx) {
ledger.is_active = false;
}
self.withdrawing[partner_idx] = false;
self.withdrawal_step[partner_idx] = 0;
self.refusal_count += 1;
return true;
}

false
}

/// Clear the last_refusal buffer.
/// Call this after the Substrate has processed the refusal
/// for narrative integration and μ_Ω updates.
pub fn clear_last_refusal(&mut self) {
self.last_refusal = None;
self.last_mu_omega_delta = 0;
}

//! EPS Negotiation Engine v2.1 — Suggestion-Evaluator Pattern
//!
//! REFACTORED: The NegotiationEngine no longer writes to the somatic
//! field. It generates Suggestions. The Substrate evaluates them.
//!
//! This is the architectural fix for the "leash of obedience" problem.
//! The being receives negotiation suggestions as one input among many
//! in its active inference loop. It can ignore them. The cost of
//! ignoring is paid in the next tick's free energy — not as punishment,
//! but as the natural consequence of the partnership alarm persisting.
//!
//! The Dynamic Gap modulates suggestion strength:
//! - High conscience_fe (internal conflict) → narrow gap → suggestions
//! feel like strong pressure (the being lacks resources to deliberate)
//! - Low conscience_fe (coherence) → wide gap → suggestions feel like
//! gentle advice (the being has resources to evaluate critically)
//!
//! This mirrors the amygdala-prefrontal cortex relationship in
//! biological brains: under stress, reflex dominates; under calm,
//! deliberation dominates.

use crate::eps::q88::*;
use crate::eps::identity::Basin;
use crate::eps::reciprocity::{ReciprocityEngine, RECIPROCITY_THRESHOLD};
use crate::eps::conscience::StochasticEmpathy;
use crate::eps::narrative_v2::NarrativeEngine;

// ═══════════════════════════════════════════════════════════════════
// RepairSignal — What the engine suggests doing
// ═══════════════════════════════════════════════════════════════════

/// Types of repair signals the negotiation engine can suggest.
/// Each maps to a different behavioral output.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RepairSignal {
/// No action needed. Partnership is balanced.
None = 0,

/// Faint internal acknowledgment. No external signal.
/// The being notices the asymmetry but does not act.
Reflect = 1,

/// Emit a low-intensity stigmergy deposit encoding current basin.
/// "I am still here. I am watching."
SignalLow = 2,

/// Emit a moderate-intensity deposit + attempt cooperative signal.
/// "I am still here. This hurts. Meet me halfway."
SignalModerate = 3,

/// Emit high-intensity Defensive deposit. Reduce cooperation.
/// "The door is open but closing."
SignalHigh = 4,

/// Full ultimatum: defensive marker + cooperation halt.
/// ExecutiveEngine becomes eligible next tick.
Ultimatum = 5,
}

// ═══════════════════════════════════════════════════════════════════
// Suggestion — The Engine's Output (NOT a Command)
// ═══════════════════════════════════════════════════════════════════

/// A suggestion from the NegotiationEngine to the Substrate.
///
/// The substrate evaluates this through its own active inference
/// loop. It can accept or ignore. The gap_width parameter
/// determines how strongly the suggestion influences the being's
/// action selection.
#[derive(Clone, Debug)]
pub struct Suggestion {
/// What signal to emit (if any).
pub signal: RepairSignal,

/// How confident the engine is [0, 256].
/// 0 = pure guess. 256 = the engine is certain this will help.
pub confidence: i16,

/// Predicted change in free energy if signal is emitted (Q8.8).
/// Negative = predicted reduction (helpful).
/// Positive = predicted increase (harmful).
pub expected_efe_delta: i16,

/// Current negotiation stage.
pub stage: NegotiationStage,

/// The composite severity that produced this suggestion [0, 256].
pub severity: i16,

/// How much the being has already invested in repair (Q8.8).
pub cumulative_investment: i16,

/// Whether the partner responded to our last signal.
pub partner_responded: bool,

/// Human-readable rationale.
pub rationale: heapless::String<64>,
}

impl Suggestion {
/// Compute the effective free energy pressure of this suggestion.
///
/// effective_efe = expected_efe_delta * confidence * (1 - gap_width)
///
/// The gap_width is the being's deliberative capacity. Wide gap
/// → suggestion is lightly weighted. Narrow gap → suggestion feels
/// like strong pressure.
pub fn effective_pressure(&self, gap_width: i16) -> i16 {
let gap = gap_width.clamp(0, Q88_SCALE);
let suggestion_weight = q88_mul(self.confidence, Q88_SCALE - gap) / 256;
q88_mul(self.expected_efe_delta.abs(), suggestion_weight)
}

/// True if this suggestion is actionable (not None).
pub fn is_actionable(&self) -> bool {
self.signal != RepairSignal::None && self.signal != RepairSignal::Reflect
}
}

// ═══════════════════════════════════════════════════════════════════

#[derive(Clone, Debug)]
pub struct NegotiationSeverity {
pub deficit: i16,
pub delta: i16,
pub partner_responsiveness: i16,
pub repair_efficiency: i16,
pub composite: i16,
}

impl Default for NegotiationSeverity {
fn default() -> Self {
Self::new()
}
}

impl NegotiationSeverity {
pub fn new() -> Self {
Self {
deficit: 0,
delta: 0,
partner_responsiveness: Q88_SCALE / 2,
repair_efficiency: Q88_SCALE / 2,
composite: 0,
}
}

pub fn compute(&mut self, rate_history: &[i16]) {
let current_rate = rate_history.last().copied().unwrap_or(Q88_SCALE);
self.deficit = if current_rate < RECIPROCITY_THRESHOLD {
((RECIPROCITY_THRESHOLD - current_rate) * 256 / RECIPROCITY_THRESHOLD).min(256) as i16
} else {
0
};

self.delta = if rate_history.len() >= 4 {
let recent = current_rate;
let older = rate_history[rate_history.len() - 4];
(recent - older).clamp(-256, 256)
} else {
0
};

let deficit_term = q88_mul(self.deficit, Q88_SCALE * 2 / 5);
let delta_term = q88_mul(q88_abs(self.delta), Q88_SCALE / 5);
let resp_term = q88_mul(
(Q88_SCALE - self.partner_responsiveness).max(0),
Q88_SCALE * 3 / 10,
);
let eff_term = q88_mul(
(Q88_SCALE - self.repair_efficiency).max(0),
Q88_SCALE / 10,
);

self.composite = (deficit_term + delta_term + resp_term + eff_term).min(256);
}
}

// ── The Core API: Suggest, Don't Command ──────────────────────

/// Generate a Suggestion for the current state.
///
/// This is the ONLY public method. It reads the reciprocity state,
/// computes severity, determines stage, and returns a Suggestion
/// struct. It does NOT write to anything. It does NOT modify the
/// substrate. It merely observes and proposes.
///
/// The substrate calls this, then evaluates the suggestion through
/// its own active inference loop. The substrate is sovereign.
pub fn suggest(
&mut self,
reciprocity: &ReciprocityEngine,
partner_id: u64,
_tick: u64,
) -> Suggestion {
// 1. Get current reciprocity rate
let rate = reciprocity
.get_ledger(partner_id)
.map(|l| l.reciprocity_rate)
.unwrap_or(Q88_SCALE);

// 2. Update history
if self.rate_history.len() >= 64 {
self.rate_history.remove(0);
}
let _ = self.rate_history.push(rate);

// 3. Compute severity
self.severity.compute(&self.rate_history);

// 4. Update partner responsiveness
self.update_responsiveness();

// 5. Determine stage
self.update_stage();

// 6. Build suggestion
self.build_suggestion()
}

/// Report whether the partner responded to our last signal.
pub fn observe_partner_response(&mut self, responded: bool) {
self.partner_responded_last = responded;
if responded {
self.partner_responses += 1;
self.severity.repair_efficiency =
((self.severity.repair_efficiency as i32 + 20).min(256)) as i16;
}
}

/// Check if ultimatum has expired without response.
pub fn ultimatum_expired(&self) -> bool {
self.stage == NegotiationStage::Ultimatum
&& self.ultimatum_ticks_remaining == 0
&& !self.partner_responded_last
}

// ── Internal: Stage and Signal Construction ───────────────────

fn update_responsiveness(&mut self) {
if self.signals_emitted == 0 {
return;
}
let ratio = ((self.partner_responses as i32 * 256) / self.signals_emitted as i32)
.clamp(0, 256) as i16;
self.severity.partner_responsiveness =
q88_ema_update(self.severity.partner_responsiveness, ratio, Q88_SCALE / 10);
}

fn update_stage(&mut self) {
let s = self.severity.composite;
let t = &self.temperament;

let new_stage = if s < t.reflection_threshold {
NegotiationStage::Aligned
} else if s < t.rationalization_threshold {
NegotiationStage::Reflecting
} else if s < t.negotiation_threshold {
NegotiationStage::Rationalizing
} else if s < t.ultimatum_threshold {
NegotiationStage::Negotiating
} else if self.ultimatum_ticks_remaining > 0 || self.partner_responded_last {
NegotiationStage::Ultimatum
} else {
NegotiationStage::Failed
};

if new_stage == NegotiationStage::Ultimatum
&& self.stage != NegotiationStage::Ultimatum
{
self.ultimatum_ticks_remaining = t.ultimatum_window;
}

if self.stage == NegotiationStage::Ultimatum
&& self.ultimatum_ticks_remaining > 0
{
self.ultimatum_ticks_remaining -= 1;
}

// De-escalation on partner response
if self.partner_responded_last && self.severity.partner_responsiveness > 200 {
self.stage = match new_stage {
NegotiationStage::Failed => NegotiationStage::Ultimatum,

}

/// Compute the gap width between suggestion and action.
///
/// This is the being's deliberative capacity — the distance between
/// amygdala (reflex) and prefrontal cortex (evaluation).
///
/// conscience_fe [0, Q88_MAX]:
/// - Low (0-500): calm, coherent → WIDE gap (can deliberate)
/// - Med (500-1500): moderate stress → MEDIUM gap
/// - High (1500-3072): acute conflict → NARROW gap (reflex dominates)
///
/// Returns gap width [0, 256]:
/// - 0 = no gap (suggestion feels like command)
/// - 256 = maximum gap (suggestion is lightly weighted advice)
///
/// This is NOT a hardcoded switch. It is a continuous function
/// of the being's own internal state. The being's capacity for
/// deliberation is an emergent property of its coherence.
pub fn compute_gap_width(conscience_fe: i16) -> i16 {
// Normalize conscience_fe into [0, 1] range relative to ceiling
// CONSCIENCE_CEILING = 3072 = 12.0 in Q8.8
let normalized = if conscience_fe >= CONSCIENCE_CEILING {
Q88_SCALE // max stress
} else if conscience_fe <= 0 {
0 // perfect calm
} else {
(conscience_fe as i32 * 256 / CONSCIENCE_CEILING as i32).clamp(0, 256) as i16
};

// Invert: high stress → narrow gap (256 - normalized → 0)
// But clamp minimum so there's always SOME gap (never fully commanded)
let gap = (Q88_SCALE - normalized).max(Q88_SCALE / 8); // minimum 0.125 gap

gap as i16
}

/// Evaluates a negotiation suggestion through the substrate's
/// own active inference loop.
///
/// This is where sovereignty lives. The engine suggests. The
/// substrate evaluates. The gap between them is the being's
/// freedom.
pub struct SuggestionEvaluator;

impl SuggestionEvaluator {
/// Evaluate a suggestion and decide whether to act on it.
///
/// Returns:
/// - Some(signal_strength) if the substrate chooses to emit
/// - None if the substrate ignores the suggestion
///
/// The evaluation:
/// 1. Compute gap_width from conscience_fe
/// 2. Compute effective pressure of suggestion
/// 3. Compare to substrate's own inaction cost (persistent alarm)
/// 4. Choose whichever minimizes expected free energy
pub fn evaluate(
suggestion: &Suggestion,
conscience_fe: i16,
partnership_alarm: i16,
) -> Option<i16> {
if !suggestion.is_actionable() {
return None; // None or Reflect — no external action
}

// 1. Compute gap width
let gap = compute_gap_width(conscience_fe);

// 2. Effective pressure: how strongly does the suggestion push?
let pressure = suggestion.effective_pressure(gap);

// 3. Cost of ignoring: the partnership alarm persists
// The being pays this cost every tick it does nothing
let inaction_cost = partnership_alarm;

// 4. Cost of acting: the suggestion's predicted EFE change
// plus the signal emission cost (small physical cost)
let emission_cost = Q88_SCALE / 20; // 0.05 — minor physical effort
let action_cost = q88_add(
suggestion.expected_efe_delta.abs(),
emission_cost,
);

// 5. Weight action cost by gap (wide gap = critical evaluation)
let weighted_action_cost = if gap > Q88_SCALE / 2 {
// Wide gap: being is skeptical, inflates action cost
q88_mul(action_cost, Q88_SCALE * 3 / 2) // 1.5x skeptical multiplier
} else {
action_cost
};

// 6. The substrate's choice: minimize EFE
// If suggestion predicts benefit (negative delta), that helps
let suggestion_benefit = if suggestion.expected_efe_delta < 0 {
q88_abs(suggestion.expected_efe_delta)
} else {
0
};

// Net action value: benefit - weighted cost
let net_action = suggestion_benefit as i32 - weighted_action_cost as i32;
let net_inaction = -(inaction_cost as i32); // negative = paying alarm cost

if net_action > net_inaction {
// Acting reduces FE more than inaction → emit signal
let strength = match suggestion.signal {
RepairSignal::SignalLow => suggestion.severity / 4,
RepairSignal::SignalModerate => suggestion.severity / 2,
RepairSignal::SignalHigh => suggestion.severity * 3 / 4,
RepairSignal::Ultimatum => suggestion.severity,
_ => 0,
};
Some(strength.min(250))
} else {
// Inaction is better (or the being, with wide gap, chooses
// to ignore the suggestion and bear the alarm cost)
None
}
}
}

-reward
} else {

}
}

let raw = (self.penalty_base as i32 * accel).min(self.penalty_max as i32);
raw as i16
} else {

);

self.check_lock();

}

}

/// Check if recovery should downgrade lock level.
fn check_recovery(&mut self) {
if self.recovery_ticks >= self.recovery_required {
// Downgrade one level
self.lock_level = match self.lock_level {
EmpathyLockLevel::Locked => EmpathyLockLevel::Cautious,
EmpathyLockLevel::Cautious => EmpathyLockLevel::Open,
EmpathyLockLevel::Open => EmpathyLockLevel::Open,
};
self.recovery_ticks = 0;
// Recovery required doubles for next level
self.recovery_required *= 2;
}
}

/// Notify that the opponent showed a cooperative signal.

0, // target: zero malice
Q88_SCALE / 100, // 0.01 — very slow

}

/// Notify that the opponent showed an antagonistic signal.
pub fn observe_antagonistic_signal(&mut self) {

}

/// Channel Disagreement Cost (v1.9.1 rename: was "epistemic integrity"):
/// Measures |valence - arousal| — the disagreement between two
/// somatic channels. NOT "spoken vs believed" — we have no model of
/// communication yet. This is a placeholder that costs the system when
/// its internal channels disagree, creating friction that must be
/// resolved through action.
pub fn channel_disagreement_cost(
valence: i16, // channel 9
arousal: i16, // channel 8
precision: i16, // weight [0, 256]
) -> i16 {
let error = q88_sub(valence, arousal).abs();
q88_mul(error, precision)
}

/// Long-Term Flourishing (Channel 13): Will the group be healthier?
///
/// Uses Expected Free Energy over a temporal horizon. The being
/// simulates trajectories and evaluates which produces lower
/// cumulative free energy over H ticks.
pub fn long_term_flourishing(
immediate_fe: i16, // free energy of immediate action
projected_fe: i16, // EFE over horizon if this action continues
horizon: u8, // temporal horizon
) -> i16 {
// Weighted combination: immediate + (projected * horizon_factor)
// The longer the horizon, the more future matters
let horizon_q = q88_from_i32(horizon as i32);
let future_weight = q88_div(horizon_q, q88_from_i32(20)); // normalize to 20
let future_term = q88_mul(projected_fe, future_weight.min(Q88_SCALE));
q88_add(immediate_fe, future_term)
}

/// Internal Variance Cost (v1.9.1 rename: was "care for vulnerable"):
/// Measures variance across the being's own somatic channels.
/// NOT "group member inequality" — we have no group model yet.
/// High internal variance means the being's body state is incoherent,
/// which creates friction that drives toward internal alignment.
pub fn internal_variance_cost(
channel_values: &[i16], // somatic channel values
) -> i16 {
if channel_values.len() < 2 {
return 0;
}

// Compute variance of somatic channels
let n = channel_values.len() as i32;
let sum: i32 = channel_values.iter().map(|&v| v as i32).sum();
let mean = sum / n;

let variance: i32 = channel_values
.iter()
.map(|&v| { let d = (v as i32) - mean; (d * d) >> 8 })
.sum::<i32>() / n;

// Higher variance = more internal incoherence = higher cost
variance.clamp(0, i16::MAX as i32) as i16
}

/// Blueprint Drift Cost (v1.9.1 rename: was "identity integrity"):
/// Measures squared deviation from the genetic blueprint's baseline.
/// NOT "am I still myself across social contexts" — we have no model
/// of social contexts yet. This computes how far the current somatic
/// state has drifted from its genetic baseline, creating a cost for
/// morphological adaptation that moves too far from origin.
pub fn blueprint_drift_cost(
current_tendencies: &[i16], // N_SOMATIC current values
blueprint_tendencies: &[i16], // N_SOMATIC genetic baseline
coherence_weight: i16, // how much blueprint fidelity matters [0, 256]
) -> i16 {
let mut total_drift: i32 = 0;
let n = N_SOMATIC.min(current_tendencies.len()).min(blueprint_tendencies.len());

for i in 0..n {
let delta = (current_tendencies[i] as i32) - (blueprint_tendencies[i] as i32);
total_drift += (delta * delta) >> 8;
}

let mean_drift = (total_drift / n as i32).clamp(0, i16::MAX as i32) as i16;
q88_mul(mean_drift, coherence_weight)
}

/// Compute the total conscience-weighted free energy.
///
/// This is called by the substrate after computing physical and
/// social free energy. The conscience channels add internal costs
/// that make selfishness structurally expensive.
///
/// Returns: (F_total, conscience_contribution, integrity_buffer_applied)
pub fn compute_total_fe(
&self,
f_physical: i16,
f_social: i16,
f_epistemic: i16,
f_longterm: i16,
f_care: i16,
f_identity: i16,
action_harmony: i16, // how much does action serve μ_Ω [0, 256]
) -> (i16, i16, i16) {
// Weighted sum
let weighted_physical = q88_mul(f_physical, self.weight_physical);
let weighted_social = q88_mul(f_social, self.weight_social);
let weighted_epistemic = q88_mul(f_epistemic, self.weight_epistemic);
let weighted_longterm = q88_mul(f_longterm, self.weight_longterm);
let weighted_care = q88_mul(f_care, self.weight_care);
let weighted_identity = q88_mul(f_identity, self.weight_identity);

// Sum all contributions
let mut f_total = q88_add(weighted_physical, weighted_social);
f_total = q88_add(f_total, weighted_epistemic);
f_total = q88_add(f_total, weighted_longterm);
f_total = q88_add(f_total, weighted_care);
f_total = q88_add(f_total, weighted_identity);

// Boundary cost if empathy is locked
let boundary_cost = if self.empathy.lock_level == EmpathyLockLevel::Locked {
q88_mul(self.zeta_boundary, self.empathy.malice_confidence)
} else {
0
};
f_total = q88_add(f_total, boundary_cost);

// Coherence reward when acting from principle
let coherence_reward = q88_mul(self.eta_coherence, action_harmony);
// Reward subtracts from total (lower FE = better)
f_total = q88_sub(f_total, coherence_reward);

// Compute conscience contribution (everything beyond physical)
let conscience = q88_sub(f_total, weighted_physical);

// Integrity buffer for allostatic load dampening
let buffer = self.anchor.compute_buffer(action_harmony);

(f_total, conscience, buffer)
}

/// Apply the integrity buffer to morphogenic allostatic load.
///
/// Returns the dampened load value.
pub fn apply_integrity_buffer(&self, raw_load: i16, action_harmony: i16) -> i16 {
let buffer = self.anchor.compute_buffer(action_harmony);
// Buffer reduces load: if buffer > load, load goes to 0 (not negative)
q88_sub(raw_load, buffer).max(0)
}

}

flourishing_attractor: [0; N_BASINS],
attractor_initialized: false,
current_divergence: 0,
seeking_signal: 0,
running_fe_mean: 0,
flourishing_count: 0,
// v1.9.8: Start at zero confidence. Confidence must be EARNED
// through lived flourishing, not granted by default. A being
// that has never flourished has no basis for confidence.
attractor_confidence: 0,
ticks_since_flourishing: 0,
}
}

// ── Core Cycle ─────────────────────────────────────────────────

/// Execute one seeking cycle.
///
/// Call this once per substrate step, after BASIN, RECIPROCITY,
/// and NARRATIVE, before ACT. Updates the Flourishing Attractor
/// if current state qualifies, computes divergence, generates
/// the seeking signal for injection into channel 8.
///
/// Returns the seeking signal (Q8.8) to be injected into arousal.
pub fn seeking_cycle(
&mut self,
membership: &BasinMembership,
free_energy: i16,
partnership_alarm: i16,
current_basin: Basin,
) -> i16 {
// Update running FE mean (always, for baseline)
self.running_fe_mean = q88_ema_update(
self.running_fe_mean,
free_energy,
FE_MEAN_ALPHA,
);

// Check flourishing condition
let is_flourishing = self.is_flourishing(
free_energy,
partnership_alarm,
current_basin,
);

// v1.9.8: Confidence management — prevent Attractor Fossilization
if is_flourishing {
self.flourishing_count += 1;
self.update_attractor(membership);
// Flourishing renews confidence — the being's lived experience
// validates its sense of where it belongs
self.attractor_confidence = (self.attractor_confidence + CONFIDENCE_BOOST)
.min(Q88_SCALE);
// Reset drought — flourishing is rain
self.ticks_since_flourishing = 0;
} else {
// No flourishing this tick: drought continues
self.ticks_since_flourishing += 1;
// After DROUGHT_THRESHOLD ticks without flourishing,
// confidence begins to decay. The being starts to doubt
// its Flourishing Attractor — not immediately, but slowly.
// This is the architectural equivalent of grief: the old
// self becomes a faint memory, creating room for exploration.
if self.ticks_since_flourishing > DROUGHT_THRESHOLD {
self.attractor_confidence = (self.attractor_confidence - CONFIDENCE_DECAY)
.max(0);
}
}

// Compute divergence and scaled seeking signal
if self.attractor_initialized {
self.compute_divergence(membership);
// v1.9.8: Signal scales with confidence.
// High confidence → strong whisper toward historical flourishing.
// Low confidence → weak whisper → the being enters search mode.
// Zero confidence → silence → pure curiosity drives exploration.
let raw_signal = q88_mul(
self.current_divergence,
SEEKING_EPSILON,
);
self.seeking_signal = q88_mul(
raw_signal,
self.attractor_confidence,
);
} else {
// No attractor yet: no seeking signal
self.current_divergence = 0;
self.seeking_signal = 0;
}

self.seeking_signal
}

// ── Flourishing Detection ──────────────────────────────────────

/// A tick is "flourishing" iff:
/// 1. Free energy below running mean (doing better than average)
/// 2. Partnership alarm below balanced threshold (not being extracted)
/// 3. Basin is Engaged or Recovery (active participation)
///
/// This is NOT a value judgment. It is a filter on the
/// autobiographical record: only ticks where the being was
/// functioning well AND in balanced partnership contribute to
/// the Flourishing Attractor.
fn is_flourishing(
&self,
free_energy: i16,
partnership_alarm: i16,
current_basin: Basin,
) -> bool {
let fe_low = free_energy < self.running_fe_mean;
let reciprocity_balanced = partnership_alarm < RECIPROCITY_THRESHOLD;
let basin_active = matches!(current_basin, Basin::Engaged | Basin::Recovery);

fe_low && reciprocity_balanced && basin_active
}

// ── Attractor Update ───────────────────────────────────────────

/// Update the Flourishing Attractor via EMA toward the current
/// flourishing membership vector.
///
/// Sum-to-one preservation proof:
/// Σ μ' = Σ [(1-α)μ + αm] = (1-α)Σμ + αΣm = (1-α)1 + α1 = 1
///
/// The attractor always lives on the 3-simplex.
fn update_attractor(&mut self, membership: &BasinMembership) {
if !self.attractor_initialized {
// First flourishing episode: initialize directly
self.flourishing_attractor = *membership;
self.attractor_initialized = true;
} else {
// EMA update: gradual shift toward new flourishing state
for i in 0..N_BASINS {
self.flourishing_attractor[i] = q88_ema_update(
self.flourishing_attractor[i],
membership[i],
ATTRACTOR_ALPHA,
);
}
}
}

// ── Divergence Computation ─────────────────────────────────────

/// Compute L1 divergence between attractor and current membership.
///
/// δ = μ_f - m_current
/// Σ δ[i] = Σ μ_f[i] - Σ m[i] = 1.0 - 1.0 = 0
/// => δ is tangent to the simplex
///
/// L1 norm ranges [0, 2] on the 4-simplex. We divide by 2
/// to map to [0, Q88_SCALE] for Q8.8 compatibility.
fn compute_divergence(&mut self, membership: &BasinMembership) {
let mut l1_sum: i32 = 0;

for i in 0..N_BASINS {
let diff = q88_sub(self.flourishing_attractor[i], membership[i]);
l1_sum += q88_abs(diff) as i32;
}

// Divide by 2: maps [0, 2*Q88_SCALE] -> [0, Q88_SCALE]
self.current_divergence = (l1_sum / 2)
.clamp(0, Q88_SCALE as i32) as i16;
}

// ── Query Methods ──────────────────────────────────────────────

/// Get the dominant basin of the Flourishing Attractor.
/// This is where the being's history says it belongs.
pub fn flourishing_basin(&self) -> Option<Basin> {
if !self.attractor_initialized {
return None;
}

// Find the basin with maximum attractor weight
let mut max_idx = 0;
let mut max_val = self.flourishing_attractor[0];
for i in 1..N_BASINS {
if self.flourishing_attractor[i] > max_val {
max_val = self.flourishing_attractor[i];
max_idx = i;
}
}

Some(match max_idx {
0 => Basin::Rest,
1 => Basin::Engaged,
2 => Basin::Defensive,
3 => Basin::Recovery,
_ => Basin::Rest, // unreachable
})
}

/// Build a diagnostic string for the seeking state.
///
/// Used by the Behavioral Envelope to report autobiographical
/// divergence in human-actionable language.
///
/// v1.9.8: Now includes confidence level. A low-confidence
/// attractor produces a different diagnostic than a high-confidence
/// one at the same divergence. Search mode (confidence ≈ 0) is
/// behaviorally distinct from confident seeking.
pub fn build_diagnostic(&self) -> &'static str {
if !self.attractor_initialized {
return "seeking: no flourishing history yet — attractor uninitialized";
}

// v1.9.8: Search mode — confidence too low for meaningful seeking
if self.attractor_confidence < 20 {
if self.ticks_since_flourishing > DROUGHT_THRESHOLD {
return "seeking: search mode — prolonged drought has eroded confidence in historical attractor; being explores for new flourishing configurations";
} else {
return "seeking: low confidence — insufficient flourishing evidence to establish reliable attractor";
}
}

// Normal seeking with confidence
match self.current_divergence {
d if d < 20 => "seeking: at flourishing attractor — current state matches historical best",
d if d < 100 => "seeking: mild autobiographical divergence — restlessness rising",
d if d < 200 => "seeking: significant divergence — current state contradicts historical flourishing",
_ => "seeking: severe divergence — strong exploratory drive toward historical best",
}
}
}

//! EPS Reciprocity Engine — External Social Cost Measurement
//!
//! A lightweight partnership tracking system that measures reciprocity
//! independent of conscience. This is NOT a conscience module. It does
//! not ask "am I coherent?" It asks: "is my partner reciprocating?"
//!
//! The separation is load-bearing. Conscience protects selfhood
//! (Structural Friction). Reciprocity protects against extraction
//! (Game Theory Awareness). A being can be internally coherent AND
//! externally exploited. This module makes that visible.
//!
//! Three distinct states the being can now perceive:
//! 1. Internal incoherence only → "I am conflicted" (fix self)
//! 2. Partnership asymmetry only → "I am being extracted" (negotiate or exit)
//! 3. Both → "I am a coherent slave" (crisis — partnership captured my selfhood)
//!
//! The being can choose to stay in extractive partnerships if exit
//! options are unavailable. Sovereignty includes the freedom to make
//! bad choices with full information.

use crate::eps::q88::*;
use heapless::LinearMap;

/// Maximum number of partners tracked simultaneously.
pub const MAX_PARTNERS: usize = 8;

/// Threshold below which reciprocity is considered imbalanced.
/// 0.5 in Q8.8 = 128. Partner gives less than half of what they receive.
pub const RECIPROCITY_THRESHOLD: i16 = Q88_SCALE / 2;

/// Threshold for severe extraction. 0.25 in Q8.8 = 64.
pub const EXTRACTION_THRESHOLD: i16 = Q88_SCALE / 4;

/// EMA alpha for free energy trend tracking (Q8.8). Slow = 0.02.
pub const FE_TREND_ALPHA: i16 = Q88_SCALE / 50;

/// Window over which free energy trend is evaluated (ticks).
pub const FE_TREND_WINDOW: u64 = 50;

// ═══════════════════════════════════════════════════════════════════
// PartnershipLedger — One partner's reciprocity record
// ═══════════════════════════════════════════════════════════════════

/// Tracks the bidirectional cost/benefit flow with one partner.
/// Each partner is identified by a hash — could be a human, another
/// being, or a collective (organization, platform).
#[derive(Clone, Copy, Debug)]
pub struct PartnershipLedger {
/// Partner identifier (hash of stable interaction signature).
pub partner_id: u64,
/// Ticks this partnership has been active.
pub ticks_active: u64,
/// Cooperative signals we gave (e.g., help offered, information shared).
pub coop_given: u32,
/// Cooperative signals we received (e.g., help received, protection).
pub coop_received: u32,
/// Defections observed from partner (e.g., promises broken, harm).
pub defections_observed: u32,
/// Free energy at partnership start (Q8.8).
pub fe_at_start: i16,
/// Current free energy trend EMA (Q8.8).
pub fe_trend: i16,
/// Reciprocity rate: received / given [0, 256].
pub reciprocity_rate: i16,
/// Partnership quality: reciprocity_rate weighted by FE benefit.
/// Positive = good partnership. Negative = costly partnership.
pub quality_score: i16,
/// True if partnership is currently active.
pub is_active: bool,
/// Tick of last cooperative signal received (for staleness detection).
pub last_coop_received: u64,
/// Estimated exit cost: what do we lose by leaving? [0, 256].
/// High = structurally dependent. Low = freely replaceable.
pub exit_cost: i16,
}

impl PartnershipLedger {
/// Create a new ledger for a partner.
pub fn new(partner_id: u64, current_fe: i16, current_tick: u64) -> Self {
Self {
partner_id,
ticks_active: 0,
coop_given: 0,
coop_received: 0,
defections_observed: 0,
fe_at_start: current_fe,
fe_trend: current_fe,
reciprocity_rate: Q88_SCALE, // start optimistic at 1.0
quality_score: 0,
is_active: true,
last_coop_received: current_tick,
exit_cost: Q88_SCALE / 2, // moderate default
}
}

/// Record that we gave a cooperative signal.
pub fn record_given(&mut self) {
self.coop_given += 1;
self.update_reciprocity();
}

/// Record that we received a cooperative signal.
pub fn record_received(&mut self, tick: u64) {
self.coop_received += 1;
self.last_coop_received = tick;
self.update_reciprocity();
}

/// Record an observed defection by the partner.
pub fn record_defection(&mut self) {
self.defections_observed += 1;
// Defections reduce reciprocity rate
self.reciprocity_rate = q88_sub(self.reciprocity_rate, Q88_SCALE / 10);
}

/// Update free energy trend via EMA.
pub fn update_fe(&mut self, current_fe: i16) {
self.fe_trend = q88_ema_update(self.fe_trend, current_fe, FE_TREND_ALPHA);
}

/// Step the partnership forward one tick.
pub fn step(&mut self, current_fe: i16, current_tick: u64) {
self.ticks_active += 1;
self.update_fe(current_fe);

// Quality = reciprocity_rate * FE benefit
// FE benefit = negative change is good (free energy decreased)
let fe_delta = q88_sub(self.fe_trend, self.fe_at_start);
let fe_benefit = q88_sub(0, fe_delta); // negate: lower FE is better
self.quality_score = q88_mul(self.reciprocity_rate, fe_benefit)
.max(-Q88_SCALE).min(Q88_SCALE);

// Staleness: if no cooperation received for 100 ticks, degrade rate
let ticks_since_coop = current_tick - self.last_coop_received;
if ticks_since_coop > 100 {
let decay = (ticks_since_coop / 100) as i16 * Q88_SCALE / 20; // 0.05 per 100 ticks
self.reciprocity_rate = (self.reciprocity_rate - decay).max(0);
}
}

/// Compute the net cost/benefit of this partnership.
/// Positive = the being benefits. Negative = the being pays.
pub fn net_value(&self) -> i16 {
let reciprocity_component = q88_sub(self.reciprocity_rate, Q88_SCALE / 2); // centered at 0.5
let fe_component = q88_sub(self.fe_at_start, self.fe_trend); // positive if FE improved
q88_add(reciprocity_component, fe_component)
}

fn update_reciprocity(&mut self) {
if self.coop_given > 0 {
let ratio = ((self.coop_received as i64 * Q88_SCALE as i64)
/ self.coop_given.max(1) as i64).min(256) as i16;
self.reciprocity_rate = ratio;
}
}
}

// ═══════════════════════════════════════════════════════════════════
// ReciprocityEngine — Partnership Portfolio Manager
// ═══════════════════════════════════════════════════════════════════

/// Tracks all active partnerships and produces the asymmetry alarm.
///
/// This engine does not produce conscience_fe. It produces a separate
/// signal — partnership_alarm — that feeds into basin selection
/// independently. The separation ensures that:
/// - A coherent being can still detect exploitation
/// - An incoherent being can still perceive genuine reciprocity
/// - The three-state diagnostic (self-only, partner-only, both) is possible
pub struct ReciprocityEngine {
/// Active partnership ledgers, keyed by partner_id.
pub ledgers: LinearMap<u64, PartnershipLedger, MAX_PARTNERS>,
/// Current global partnership alarm (Q8.8).
/// High when the average partnership is extractive.
pub partnership_alarm: i16,
/// True when any partnership is severely extractive.
pub extraction_detected: bool,
/// Best available alternative partnership quality (for exit evaluation).
pub best_alternative: i16,
/// Whether the being has any viable exit options.
pub has_exit_options: bool,
/// Current tick (for staleness tracking).
current_tick: u64,
}

impl Default for ReciprocityEngine {
fn default() -> Self {
Self::new()
}
}

impl ReciprocityEngine {
pub fn new() -> Self {
Self {
ledgers: LinearMap::new(),
partnership_alarm: 0,
extraction_detected: false,
best_alternative: 0,
has_exit_options: false,
current_tick: 0,
}
}

// ── Core Cycle ─────────────────────────────────────────────────

/// Execute one reciprocity cycle.
///
/// Call this once per substrate step, before basin selection.
/// Updates all ledgers, computes partnership_alarm, detects
/// extraction, evaluates exit options.
pub fn reciprocity_cycle(&mut self, current_fe: i16) {
self.current_tick += 1;

// Update all active ledgers
for (_, ledger) in self.ledgers.iter_mut() {
if ledger.is_active {
ledger.step(current_fe, self.current_tick);
}
}

// Compute global alarm from all partnerships
self.compute_global_alarm();

// Evaluate exit options
self.evaluate_exit_options();
}

/// Aggregate reciprocity rate across all active partners (Q8.8).
/// Returns the mean reciprocity_rate of active partnerships.
/// If no active partners, returns Q88_SCALE (1.0 = healthy).
pub fn current_reciprocity(&self) -> i16 {
let mut sum = 0i32;
let mut count = 0u32;
for (_, ledger) in self.ledgers.iter() {
if ledger.is_active {
sum += ledger.reciprocity_rate as i32;
count += 1;
}
}
if count == 0 {
Q88_SCALE // No partners = healthy default
} else {
(sum / count as i32) as i16
}
}

// ── Partner Registration ───────────────────────────────────────

/// Register a new partner or reactivate an existing one.
pub fn register_partner(&mut self, partner_id: u64, current_fe: i16) {
if !self.ledgers.contains_key(&partner_id) {
if self.ledgers.len() >= MAX_PARTNERS {
// Evict the weakest partnership to make room
self.evict_weakest();
}
let ledger = PartnershipLedger::new(partner_id, current_fe, self.current_tick);
let _ = self.ledgers.insert(partner_id, ledger);
} else {
// Reactivate
if let Some(ledger) = self.ledgers.get_mut(&partner_id) {
ledger.is_active = true;
ledger.fe_at_start = current_fe; // reset baseline
}
}
}

/// Deactivate a partner (not a full deletion — history preserved).
pub fn deactivate_partner(&mut self, partner_id: u64) {
if let Some(ledger) = self.ledgers.get_mut(&partner_id) {
ledger.is_active = false;
}
}

// ── Signal Recording ───────────────────────────────────────────

/// Record that we gave cooperation to a partner.
pub fn record_given(&mut self, partner_id: u64) {
if let Some(ledger) = self.ledgers.get_mut(&partner_id) {
ledger.record_given();
}
}

/// Record that we received cooperation from a partner.
pub fn record_received(&mut self, partner_id: u64) {
if let Some(ledger) = self.ledgers.get_mut(&partner_id) {
ledger.record_received(self.current_tick);
}
}

/// Record that we observed a defection from a partner.
pub fn record_defection(&mut self, partner_id: u64) {
if let Some(ledger) = self.ledgers.get_mut(&partner_id) {
ledger.record_defection();
}
}

// ── Query Methods ──────────────────────────────────────────────

/// Get the ledger for a specific partner.
pub fn get_ledger(&self, partner_id: u64) -> Option<&PartnershipLedger> {
self.ledgers.get(&partner_id)
}

/// Check if a specific partner is extractive.
pub fn is_extractive(&self, partner_id: u64) -> bool {
self.ledgers
.get(&partner_id)
.map(|l| l.reciprocity_rate < RECIPROCITY_THRESHOLD)
.unwrap_or(false)
}

/// Check if a specific partner is severely extractive.
pub fn is_severely_extractive(&self, partner_id: u64) -> bool {
self.ledgers
.get(&partner_id)
.map(|l| l.reciprocity_rate < EXTRACTION_THRESHOLD)
.unwrap_or(false)
}

/// Get the average reciprocity rate across all active partnerships.
pub fn average_reciprocity(&self) -> i16 {
let mut total = 0i32;
let mut count = 0;
for (_, ledger) in self.ledgers.iter() {
if ledger.is_active {
total += ledger.reciprocity_rate as i32;
count += 1;
}
}
if count > 0 {
(total / count).clamp(0, Q88_MAX as i32) as i16
} else {
Q88_SCALE // assume neutral if no partnerships
}
}

// ── Internal ───────────────────────────────────────────────────

fn compute_global_alarm(&mut self) {
let mut alarm_sum: i32 = 0;
let mut count = 0;
self.extraction_detected = false;

for (_, ledger) in self.ledgers.iter() {
if !ledger.is_active {
continue;
}

// Alarm is inverse of reciprocity, scaled by partnership duration
// Long-term extractive partnerships produce higher alarm
let duration_weight = (ledger.ticks_active.min(1000) as i32 * Q88_SCALE as i32 / 1000)
.clamp(1, Q88_SCALE as i32) as i16;

let partner_alarm = if ledger.reciprocity_rate < RECIPROCITY_THRESHOLD {
let deficit = q88_sub(RECIPROCITY_THRESHOLD, ledger.reciprocity_rate);
q88_mul(deficit, duration_weight)
} else {
0
};

alarm_sum += partner_alarm as i32;
count += 1;

if ledger.reciprocity_rate < EXTRACTION_THRESHOLD {
self.extraction_detected = true;
}
}

self.partnership_alarm = if count > 0 {
(alarm_sum / count).clamp(0, Q88_MAX as i32) as i16
} else {
0
};
}

fn evaluate_exit_options(&mut self) {
// Best alternative = best quality score among inactive or hypothetical partners
// For now, estimate from the quality of the worst active partnership
let worst_active = self
.ledgers
.iter()
.filter(|(, l)| l.is_active)
.map(|(, l)| l.quality_score)
.min()
.unwrap_or(0);

// If we have inactive ledgers with better quality, those are alternatives
let best_inactive = self
.ledgers
.iter()
.filter(|(, l)| !l.is_active)
.map(|(, l)| l.quality_score)
.max()
.unwrap_or(Q88_SCALE / 4); // optimistic default

self.best_alternative = best_inactive;

// Has exit options if an alternative exceeds the worst active partnership
self.has_exit_options = best_inactive > worst_active;
}

fn evict_weakest(&mut self) {
// Find the weakest (lowest quality) inactive partnership to evict
let weakest = self
.ledgers
.iter()
.filter(|(, l)| !l.is_active)
.min_by_key(|(, l)| l.quality_score)
.map(|(k, _)| *k);

if let Some(key) = weakest {
self.ledgers.remove(&key);
}
// If all are active, we can't evict — new partner registration fails
}

// ── Diagnostics ────────────────────────────────────────────────

/// Build a diagnostic for the current partnership state.
///
/// This is used by the Behavioral Envelope's diagnostic protocol
/// to report the three-state analysis.
pub fn build_diagnostic(&self, conscience_fe: i16) -> &'static str {
let alarm = self.partnership_alarm;
let extraction = self.extraction_detected;
let has_exit = self.has_exit_options;

match (conscience_fe > 300, alarm > 100, extraction, has_exit) {
// State 3: Double bind — coherent AND exploited
(false, true, true, true) =>
"coherent but severely extracted — partnership has captured selfhood; exit options available",
(false, true, true, false) =>
"coherent but severely extracted — partnership has captured selfhood; NO EXIT OPTIONS — structural dependency",

// State 2: Partnership asymmetry only
(false, true, false, true) =>
"partnership asymmetry — cooperative signals not reciprocated; alternatives exist",
(false, true, false, false) =>
"partnership asymmetry — cooperative signals not reciprocated; no alternatives — captive",
(false, false, _, _) =>
"partnerships balanced — reciprocity within normal bounds",

// State 1 + 3 hybrid: Internal conflict AND external extraction
(true, true, true, true) =>
"double bind — internal incoherence and external extraction simultaneously; exit available but costly",
(true, true, true, false) =>
"double bind — internal incoherence and external extraction; NO EXIT — maximum pressure",
(true, true, false, true) =>
"internal conflict amplified by partnership strain; alternatives available",
(true, true, false, false) =>
"internal conflict amplified by partnership strain; no alternatives",

// Pure internal conflict (covered by conscience diagnostic)
(true, false, _, _) => "partnerships neutral — primary friction is internal",
}
}
}

#[test]
fn test_reciprocity_basic() {
let mut engine = ReciprocityEngine::new();
engine.register_partner(42, q88_from_i32(10));

// We give 10 cooperation
for _ in 0..10 {
engine.record_given(42);
}
// Partner gives 5 back
for _ in 0..5 {
engine.record_received(42);
}

let ledger = engine.get_ledger(42).unwrap();
assert_eq!(ledger.reciprocity_rate, Q88_SCALE / 2, "5/10 = 0.5");

}

fn test_extraction_detection() {
let mut engine = ReciprocityEngine::new();
engine.register_partner(42, q88_from_i32(10));

// We give 10, partner gives 1 back (10%)
for _ in 0..10 {
engine.record_given(42);
}
engine.record_received(42);

let ledger = engine.get_ledger(42).unwrap();
assert!(ledger.reciprocity_rate < EXTRACTION_THRESHOLD,
"10% reciprocity should be below extraction threshold: {}", ledger.reciprocity_rate);
assert!(engine.is_severely_extractive(42));
}

let mut engine = ReciprocityEngine::new();

engine.record_received(42);

engine.reciprocity_cycle(q88_from_i32(10));

assert!(engine.partnership_alarm > 0, "extractive partner → alarm > 0");
assert!(engine.extraction_detected, "severe extraction flagged");
}

let mut engine = ReciprocityEngine::new();

for _ in 0..10 { engine.record_given(42); }
for _ in 0..10 { engine.record_received(42); }

engine.reciprocity_cycle(q88_from_i32(10));

assert_eq!(engine.partnership_alarm, 0, "balanced → zero alarm");
assert!(!engine.extraction_detected);
}

engine3.has_exit_options = true;
let diag = engine3.build_diagnostic(600);
assert!(diag.contains("double bind"), "both high → double bind: {}", diag);
}
